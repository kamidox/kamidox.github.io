DQN - Deep Q-Network

一、Q-Learning 的瓶颈

问题 原因
Q 表存不下 状态×动作空间一大，表就爆炸
不能处理连续状态 CartPole 观测是连续值，Taxi 是离散的
泛化能力差 没见过的状态，Q 表里没有，就瞎猜

DQN 的解法：用神经网络代替 Q 表。

Q-Learning:  Q[state][action] = 8.5
DQN:         Q_value = neural_network(state) → [Q(a=0), Q(a=1), ...]

输入一个状态，网络直接输出所有动作的 Q 值。

二、DQN 三大核心创新

1️⃣ Experience Replay（经验回放）

Q-Learning 的问题：每次拿当前一步的经验立刻更新，样本之间高度相关（连续帧几乎一样），训练不稳定。

DQN 的做法：把每一步的 (s, a, r, s', done) 存进一个回放缓冲区（Replay Buffer），训练时随机采样一个 batch 来更新。

Replay Buffer（容量 10000）
┌──────────────────────────────────┐
│ (s₁, a₁, r₁, s₁', done₁)        │
│ (s₂, a₂, r₂, s₂', done₂)        │
│ ...                              │
│ (s₁₀₀₀₀, a, r, s', done)        │
└──────────────────────────────────┘
         ↓ 随机采样 batch=64
         ↓ 用来更新网络

好处： - 打破时间相关性 - 同一份经验可以反复利用 - 数据分布更稳定

2️⃣ Target Network（目标网络）

问题：Q-Learning 更新公式里，max Q(s', a') 用的就是正在训练的网络。网络一变，目标也跟着变，训练像”追着一个移动的目标跑”。

DQN 的做法：维护两个网络：

网络 作用
Policy Network（在线网络） 选动作、被梯度更新
Target Network（目标网络） 提供 TD Target 里的 max Q(s', a')

Target Network 每隔 C 步才从 Policy Network 拷贝一次权重。

TD Target = r + γ · max Q_target(s', a')
                        ↑ 固定一段时间不变

3️⃣ ε-Greedy 探索（和 Q-Learning 一样）

三、DQN 算法流程 

初始化 policy_net 和 target_net（权重相同）
初始化 replay_buffer

for episode = 1 to N:
    重置环境，得到 s
    for step = 1 to max_steps:
        用 ε-greedy 选动作 a
        执行 a，得到 (s, a, r, s', done)
        存入 replay_buffer

        # 从 buffer 采样训练
        if buffer 够大了:
            采样 batch 个 (s, a, r, s', done)
            用 policy_net 算 Q(s, a)
            用 target_net 算 max Q(s', a')
            TD Target = r + γ · max Q(s', a') · (1 - done)
            最小化 MSE Loss: (Q(s,a) - Target)²
            更新 policy_net

        # 定期同步
        if step % C == 0:
            target_net.load_state_dict(policy_net.state_dict())

四、完整 Python 实现（CartPole-v1）

依赖 

uv pip install torch gymnasium "gymnasium[classic-control]" numpy matplotlib

完整代码 

"""
DQN on CartPole-v1
Python 3.12 + PyTorch + Gymnasium
"""

import gymnasium as gym
import torch
import torch.nn as nn
import torch.optim as optim
import numpy as np
import random
from collections import deque, namedtuple
from itertools import count
import matplotlib.pyplot as plt

# ==================== 超参数 ====================
GAMMA = 0.99              # 折扣因子
EPSILON_START = 1.0       # 初始探索率
EPSILON_END = 0.05        # 最小探索率
EPSILON_DECAY = 1000      # 衰减步数（线性衰减）
BATCH_SIZE = 64            # 采样批量
MEMORY_SIZE = 10000        # 回放缓冲区大小
TARGET_UPDATE = 10         # 每 10 个 episode 同步一次 target net
LR = 1e-3                  # 学习率
NUM_EPISODES = 500         # 训练回合数
MAX_STEPS = 500            # 每回合最大步数

# 经验元组
Transition = namedtuple("Transition",
                         ["state", "action", "reward", "next_state", "done"])


# ==================== Q 网络 ====================
class DQN(nn.Module):
    """简单的 3 层全连接网络"""
    def __init__(self, state_dim, action_dim):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(state_dim, 128),
            nn.ReLU(),
            nn.Linear(128, 128),
            nn.ReLU(),
            nn.Linear(128, action_dim),
        )

    def forward(self, x):
        return self.net(x)


# ==================== 经验回放缓冲区 ====================
class ReplayBuffer:
    def __init__(self, capacity):
        self.buffer = deque(maxlen=capacity)

    def push(self, *args):
        self.buffer.append(Transition(*args))

    def sample(self, batch_size):
        return random.sample(self.buffer, batch_size)

    def __len__(self):
        return len(self.buffer)


# ==================== ε-贪心策略 ====================
def select_action(state, policy_net, epsilon):
    """ε-贪心：以 ε 概率随机探索，否则选 Q 值最大的动作"""
    if random.random() < epsilon:
        return random.randint(0, env.action_space.n - 1)
    else:
        with torch.no_grad():
            state_t = torch.tensor(state, dtype=torch.float32).unsqueeze(0)
            q_values = policy_net(state_t)
            return q_values.argmax().item()


# ==================== 训练一步 ====================
def optimize_model(policy_net, target_net, optimizer, memory):
    if len(memory) < BATCH_SIZE:
        return

    transitions = memory.sample(BATCH_SIZE)
    batch = Transition(*zip(*transitions))

    # 转成 tensor
    state_batch = torch.tensor(batch.state, dtype=torch.float32)
    action_batch = torch.tensor(batch.action, dtype=torch.int64).unsqueeze(1)
    reward_batch = torch.tensor(batch.reward, dtype=torch.float32)
    next_state_batch = torch.tensor(batch.next_state, dtype=torch.float32)
    done_batch = torch.tensor(batch.done, dtype=torch.float32)

    # Q(s, a) — policy_net 对"已选动作"的 Q 值
    q_values = policy_net(state_batch)           # (64, action_dim)
    q_values = q_values.gather(1, action_batch)  # (64, 1)
    q_values = q_values.squeeze(1)               # (64,)

    # max Q(s', a') — target_net 算下一个状态的最大 Q
    with torch.no_grad():
        next_q_values = target_net(next_state_batch).max(1)[0]  # (64,)

    # TD Target: r + γ · max Q(s', a') · (1 - done)
    target = reward_batch + GAMMA * next_q_values * (1 - done_batch)

    # 均方误差损失
    loss = nn.MSELoss()(q_values, target)

    # 反向传播
    optimizer.zero_grad()
    loss.backward()
    optimizer.step()


# ==================== ε 衰减 ====================
def decay_epsilon(step):
    """线性衰减：从 EPSILON_START 到 EPSILON_END"""
    return EPSILON_END + (EPSILON_START - EPSILON_END) * \
           np.exp(-step / EPSILON_DECAY)


# ==================== 主训练 ====================
env = gym.make("CartPole-v1")

state_dim = env.observation_space.shape[0]   # 4
action_dim = env.action_space.n              # 2

policy_net = DQN(state_dim, action_dim)
target_net = DQN(state_dim, action_dim)
target_net.load_state_dict(policy_net.state_dict())
target_net.eval()  # target net 不训练

optimizer = optim.Adam(policy_net.parameters(), lr=LR)
memory = ReplayBuffer(MEMORY_SIZE)

episode_rewards = []
global_step = 0

print("🚀 开始训练 DQN...\n")

for episode in range(NUM_EPISODES):
    state, _ = env.reset()
    total_reward = 0

    for step in range(MAX_STEPS):
        epsilon = decay_epsilon(global_step)
        action = select_action(state, policy_net, epsilon)

        next_state, reward, terminated, truncated, _ = env.step(action)
        done = terminated or truncated

        memory.push(state, action, reward, next_state, done)

        optimize_model(policy_net, target_net, optimizer, memory)

        state = next_state
        total_reward += reward
        global_step += 1

        if done:
            break

    # 定期同步 target net
    if episode % TARGET_UPDATE == 0:
        target_net.load_state_dict(policy_net.state_dict())

    episode_rewards.append(total_reward)

    # 打印进度
    if (episode + 1) % 50 == 0:
        avg_reward = np.mean(episode_rewards[-50:])
        print(f"Episode {episode+1:3d} | "
              f"Avg Reward (last 50): {avg_reward:6.1f} | "
              f"Epsilon: {epsilon:.3f} | "
              f"Buffer: {len(memory)}")

env.close()
print("\n✅ 训练完成！")


# ==================== 可视化 ====================
plt.figure(figsize=(10, 5))
plt.plot(episode_rewards, alpha=0.3, label="Raw")

# 滑动平均
window = 20
smoothed = np.convolve(episode_rewards,
                        np.ones(window)/window, mode="valid")
plt.plot(range(window-1, len(episode_rewards)),
         smoothed, label=f"Moving Avg ({window})", linewidth=2)

plt.xlabel("Episode")
plt.ylabel("Total Reward")
plt.title("DQN Training on CartPole-v1")
plt.legend()
plt.grid(True, alpha=0.3)
plt.show()


# ==================== 测试（无探索） ====================
print("\n🎮 开始测试（纯贪心，ε=0）...")
test_env = gym.make("CartPole-v1", render_mode="human")

for test_ep in range(3):
    state, _ = test_env.reset()
    total_reward = 0
    done = False

    while not done:
        with torch.no_grad():
            state_t = torch.tensor(state, dtype=torch.float32).unsqueeze(0)
            action = policy_net(state_t).argmax().item()
        state, reward, terminated, truncated, _ = test_env.step(action)
        total_reward += reward
        done = terminated or truncated

    print(f"Test Episode {test_ep+1}: Total Reward = {total_reward}")

test_env.close()

六、DQN vs Q-Learning 对比

Q-Learning DQN
Q 值存储 表格 神经网络
状态空间 只能离散 连续/离散都行
泛化 无 有（网络参数共享）
样本效率 低（用一次就丢） 高（经验回放复用）
训练稳定性 稳定 需要 Target Network
代码复杂度 简单 中等

七、DQN 的后续改进（知道就好）

改进 解决什么问题
Double DQN Target 高估 Q 值
Dueling DQN 把 Q 拆成 V + A，更好估计状态价值
Prioritized Replay 重要经验多采样
Noisy Nets 用参数噪声代替 ε-greedy
Rainbow 以上全用上，SOTA

八、关键概念速查

概念 一句话
Q Network 输入状态，输出每个动作的 Q 值
Replay Buffer 经验池，随机采样打破相关性
Target Network 提供稳定的 TD Target
TD Error Q(s,a) 和 r + γ·max Q(s',a') 的差
Off-policy 用 ε-greedy 探索的数据来训练贪心策略
